--create the trigger
CREATE TRIGGER [dba_trigger_log_instance_events]
ON ALL SERVER

FOR   CREATE_DATABASE, DROP_DATABASE, ALTER_DATABASE, CREATE_LOGIN, ALTER_LOGIN,

      DROP_LOGIN, GRANT_SERVER, DENY_SERVER, REVOKE_SERVER, CREATE_TRIGGER, DROP_TRIGGER, ADD_ROLE_MEMBER, DROP_ROLE_MEMBER, ADD_SERVER_ROLE_MEMBER, DROP_SERVER_ROLE_MEMBER, CREATE_USER, DROP_USER, ALTER_USER
AS

set ansi_padding on
set ansi_warnings on
set CONCAT_NULL_YIELDS_NULL on
set ARITHABORT on

set nocount on
declare @data xml
set @data = EVENTDATA()

--pull free space from DBA DB
declare @dba_db_available_space_mb decimal(18,2)

EXEC sp_executesql N'USE DBA;
SELECT    @dba_db_available_space_mb_output = CONVERT(DECIMAL(18,2),SUM(size/128.0 - CAST(FILEPROPERTY(name, ''SpaceUsed'') AS int)/128.0))
FROM sys.database_files
WHERE type_desc = ''ROWS''',
N'@dba_db_available_space_mb_output decimal(18,2) OUTPUT',
@dba_db_available_space_mb_output = @dba_db_available_space_mb OUTPUT

/*
SELECT @dba_db_available_space_mb,
    @dba_db_available_space_mb*1024.0 [dba_db_available_space_kb]
*/

IF EXISTS (select [name] from DBA.sys.tables where name = 'DBA_Instance_DDL_Trigger_Event_Information')
BEGIN
    IF @dba_db_available_space_mb > 0.02
    BEGIN

        INSERT INTO [DBA].[dbo].[DBA_Instance_DDL_Trigger_Event_Information](

              DatabaseName,

              EventType,

              ObjectName,

              ObjectType,

              SqlCommand,

              LoginName,

              HostName,

              AppName)
        VALUES(
              ISNULL(@data.value('(/EVENT_INSTANCE/DatabaseName)[1]', 'varchar(256)'), 'N/A'),
              ISNULL(@data.value('(/EVENT_INSTANCE/EventType)[1]', 'varchar(50)'), 'N/A'),
              ISNULL(@data.value('(/EVENT_INSTANCE/ObjectName)[1]', 'varchar(256)'), 'N/A'),
              ISNULL(@data.value('(/EVENT_INSTANCE/ObjectType)[1]', 'varchar(25)'), 'N/A'),
              ISNULL(@data.value('(/EVENT_INSTANCE/TSQLCommand)[1]', 'varchar(8000)'), 'N/A'),
              ISNULL(@data.value('(/EVENT_INSTANCE/LoginName)[1]', 'varchar(256)'),'N/A'),
              ISNULL(Host_Name(), 'N/A'),
              ISNULL(App_Name(), 'N/A')
              )
    END
END

GO

SET ANSI_NULLS OFF
GO

SET QUOTED_IDENTIFIER OFF
GO

ENABLE TRIGGER [dba_trigger_log_instance_events] ON ALL SERVER
GO
