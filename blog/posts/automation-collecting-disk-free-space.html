Automation --Collecting Disk free space

Automation
SELECT 'Start - DBA_Maint_Collect_Disk_Free_Space deployment'
GO

USE [DBA]
GO

/*** Drop and re-create tables ***/

IF EXISTS (select name from sys.tables where name = 'DBA_DiskFreeSpace')
BEGIN
 DROP TABLE [dbo].[DBA_DiskFreeSpace];
END

CREATE TABLE [dbo].[DBA_DiskFreeSpace](
 [driveLetter] [varchar](255) NOT NULL,
 [capacityGB] [decimal](16, 2) NULL,
 [freeGB] [decimal](16, 2) NULL,
 [insertdate] [datetime] NULL,
 [percentFreeSpace] [decimal](16, 2) NULL
) ON [PRIMARY]
GO

IF EXISTS (select name from sys.tables where name = 'DBA_DiskFreeSpace_Archive')
BEGIN
 DROP TABLE [dbo].[DBA_DiskFreeSpace_Archive];
END

CREATE TABLE [dbo].[DBA_DiskFreeSpace_Archive](
 [driveLetter] [varchar](255) NOT NULL,
 [capacityGB] [decimal](16, 2) NULL,
 [freeGB] [decimal](16, 2) NULL,
 [insertdate] [datetime] NULL,
 [percentFreeSpace] [decimal](16, 2) NULL
) ON [PRIMARY]

GO

CREATE CLUSTERED INDEX [idx_clu_insertdate] ON [dbo].[DBA_DiskFreeSpace_Archive]
(
 [insertdate] ASC
)WITH (PAD_INDEX = OFF, STATISTICS_NORECOMPUTE = OFF, SORT_IN_TEMPDB = OFF, DROP_EXISTING = OFF, ONLINE = OFF, ALLOW_ROW_LOCKS = ON, ALLOW_PAGE_LOCKS = ON, FILLFACTOR = 100)
GO

USE [msdb]
GO

IF EXISTS (select name from dbo.sysjobs where name = 'DBA_Maint_Collect_Disk_Free_Space')
BEGIN
 EXEC dbo.sp_delete_job @job_name = 'DBA_Maint_Collect_Disk_Free_Space';
END
GO

/****** Object:  Job [DBA_Maint_Collect_Disk_Free_Space]    Script Date: 7/13/2015 11:38:28 AM ******/
BEGIN TRANSACTION
DECLARE @ReturnCode INT
SELECT @ReturnCode = 0

BEGIN
EXEC @ReturnCode = msdb.dbo.sp_add_category @class=N'JOB', @type=N'LOCAL', @name=N'[Uncategorized (Local)]'

END

--skip if sql 2005
IF (SELECT SUBSTRING(convert(sysname, SERVERPROPERTY('productversion')), 1, charindex('.',convert(sysname, SERVERPROPERTY('productversion')))-1)) = 9 GOTO QuitWithRollback

DECLARE @jobId BINARY(16)

EXEC @ReturnCode = msdb.dbo.sp_add_jobstep @job_id=@jobId, @step_name=N'Archive Data from Last Pull',
  @step_id=1,
  @cmdexec_success_code=0,
  @on_success_action=3,

INSERT INTO [dbo].[DBA_DiskFreeSpace_Archive]

/****** Object:  Step [Pull data via Powershell]    Script Date: 7/13/2015 11:38:29 AM ******/

  @step_id=3,

  @command=N'#data table to hold results
Function out-DataTable
{
  $dt = new-object Data.datatable
  $First = $true

  foreach ($item in $input){
    $DR = $DT.NewRow()
    $Item.PsObject.get_properties() | foreach {
      if ($first) {
        $Col =  new-object Data.DataColumn
        $Col.ColumnName = $_.Name.ToString()
        $DT.Columns.Add($Col)       }
      if ($_.value -eq $null) {
        $DR.Item($_.Name) = "[empty]"
      }
      elseif ($_.IsArray) {
        $DR.Item($_.Name) =[string]::Join($_.value ,";")
      }
      else {
        $DR.Item($_.Name) = $_.value
      }
    }
    $DT.Rows.Add($DR)
    $First = $false
  }

  return @(,($dt))

}

#function to retrieve disk information
Function Get-DisksSpace ([string]$Servername, $unit= "GB")
{
$measure = "1$unit"

$dt = get-date

Get-WmiObject -computername $serverName -query "
select Name, FreeSpace, Capacity
  from Win32_Volume
 where (DriveType = 2 or DriveType = 3) and Capacity IS NOT NULL" `
| select @{Label="driveLetter"; Expression={$_.Name}} `
        , @{Label="capacity$unit";Expression={"{0:n2}" -f($_.Capacity/$measure)}} `
        , @{Label="free$unit";Expression={"{0:n2}" -f($_.freespace/$measure)}} `
       , @{Label="insertdate"; Expression={"$dt"}} `
        , @{Label="percentFreeSpace";Expression={"{0:n2}" -f($_.freespace/$_.Capacity*100.0)}} }

#execute the functions

$serverInstance = "$(ESCAPE_DQUOTE(SRVR))"
#$serverInstance = "PACDCSQL03"
$s = "$env:computername"
Get-DisksSpace $s

$dataTable = Get-DisksSpace $s | out-DataTable
$connectionString = "Data Source=$serverInstance; Integrated Security=True;Initial Catalog=DBA;"
$bulkCopy = new-object ("Data.SqlClient.SqlBulkCopy") $connectionString
$bulkCopy.DestinationTableName = "dbo.DBA_DiskFreeSpace"
$bulkCopy.WriteToServer($dataTable)
',
  @database_name=N'master',
  @flags=0
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
