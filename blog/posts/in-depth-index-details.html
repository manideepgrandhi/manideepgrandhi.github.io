In depth index details

index
SELECT [index_id] AS [ID],
OBJECT_ID AS ObjectId
        ,Object_Name(Object_id) AS ObjectName
 , [index_depth] AS [D],Index_Type_Desc AS IndexDescription
    , [index_level] AS [L]
    , [record_count] AS [Rows]
    , [page_count] AS [Pages]
    , [avg_page_space_used_in_percent] AS [Page:Percent Full]
    , [min_record_size_in_bytes] AS [Row:MinLen]
    , [max_record_size_in_bytes] AS [Row:MaxLen]
    , [avg_record_size_in_bytes] AS [Row:AvgLen]
FROM [sys].[dm_db_index_physical_stats]
    (DB_ID (N'AdventureWorksDW2012')     -- Database ID
    , OBJECT_ID (N'AdventureWorksDW2012.dbo.DimCustomer') -- Object ID
    , 1           -- Index ID
    , NULL           -- Partition ID
    , 'DETAILED');         -- Mode
GO
SELECT [DatabaseName]
    ,[ObjectId]
    ,[ObjectName],record_count,[page_count]
    ,[IndexId],[index_level]
    ,[IndexDescription]
    ,CONVERT(DECIMAL(16, 1), (SUM([avg_record_size_in_bytes] * [record_count]) / (1024.0 * 1024))) AS [IndexSize(MB)]
    ,[lastupdated] AS [StatisticLastUpdated]
    ,[AvgFragmentationInPercent]
FROM (
    SELECT DISTINCT DB_Name(Database_id) AS 'DatabaseName'
        ,OBJECT_ID AS ObjectId
        ,Object_Name(Object_id) AS ObjectName
        ,Index_ID AS IndexId
        ,Index_Type_Desc AS IndexDescription
        ,avg_record_size_in_bytes
        ,record_count,[page_count],[index_level]
        ,STATS_DATE(object_id, index_id) AS 'lastupdated'
        ,CONVERT([varchar](512), round(Avg_Fragmentation_In_Percent, 3)) AS 'AvgFragmentationInPercent'
    FROM sys.dm_db_index_physical_stats
  (DB_ID (N'AdventureWorksDW2012')     -- Database ID
    , OBJECT_ID (N'AdventureWorksDW2012.dbo.DimCustomer') -- Object ID
    , null        -- Index ID
    , NULL           -- Partition ID
    , 'DETAILED')
    --WHERE OBJECT_ID IS NOT NULL
        --AND Avg_Fragmentation_In_Percent <> 0
    ) T
GROUP BY DatabaseName
    ,ObjectId
    ,ObjectName
    ,IndexId
    ,IndexDescription,
 index_level
    ,lastupdated,
 [page_count],
 record_count
    ,AvgFragmentationInPercent
