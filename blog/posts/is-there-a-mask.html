is there a mask

mask
SELECT obj.name AS ObjName,
       c.name,
       c.is_masked,
       c.masking_function
FROM sys.masked_columns AS c
    INNER JOIN sys.all_objects AS obj
        ON c.[object_id] = obj.[object_id]
WHERE c.is_masked = 1
	AND obj.is_ms_shipped = 0;
