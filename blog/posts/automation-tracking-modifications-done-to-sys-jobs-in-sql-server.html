Automation-Tracking modifications done to Sys jobs in SQL Server

Automation
/***** DEPLOY sysjobs and sysjobsteps audit logging *****/

SELECT 'Start - sysjobs and sysjobsteps audit logging deployment'
GO

/***** dbo.sysjobs *****/

USE [DBA]
GO

/*create the logging table in DBA database*/

IF NOT EXISTS (select 1 from DBA.sys.objects where name = 'Logging_msdb_sysjobs_modifications')
BEGIN

 CREATE TABLE [dbo].[Logging_msdb_sysjobs_modifications](
  [job_id] [uniqueidentifier] NOT NULL,
  [originating_server_id] [int] NOT NULL,
  [name] [sysname] NOT NULL,
  [enabled] [tinyint] NOT NULL,
  [description] [nvarchar](512) NULL,
  [start_step_id] [int] NOT NULL,
  [category_id] [int] NOT NULL,
  [owner_sid] [varbinary](85) NOT NULL,
  [notify_level_eventlog] [int] NOT NULL,
  [notify_level_email] [int] NOT NULL,
  [notify_level_netsend] [int] NOT NULL,
  [notify_level_page] [int] NOT NULL,
  [notify_email_operator_id] [int] NOT NULL,
  [notify_netsend_operator_id] [int] NOT NULL,
  [notify_page_operator_id] [int] NOT NULL,
  [delete_level] [int] NOT NULL,
  [date_created] [datetime] NOT NULL,
  [date_modified] [datetime] NOT NULL,
  [version_number] [int] NOT NULL,
  [host_name] varchar(255) NULL,
  [login_name] varchar(1000) NULL,
  [modification_type] varchar(50) NULL,
  [insert_date_time] datetime NULL
 ) ON [PRIMARY];

 ALTER TABLE dbo.[Logging_msdb_sysjobs_modifications] ADD CONSTRAINT
  DF_Logging_sysjobs_modifications_insert_date_time_1 DEFAULT getdate() FOR insert_date_time;
END
GO

/*** CREATE TRIGGERS ***/

/* LOG INSERTS */
USE msdb
GO

-- =============================================
-- Author:  S. Kusen
-- Create date: 2014-08-01
-- Description: This will log inserts to the sysjobs table and who did it, and when.
-- =============================================
IF EXISTS (select 1 from sys.triggers where name = 'comcast_dba_sysjobs_insert_log_entry')
EXEC('DROP TRIGGER comcast_dba_sysjobs_insert_log_entry');

IF NOT EXISTS (select 1 from sys.triggers where name = 'comcast_dba_sysjobs_insert_log_entry')
EXEC('CREATE TRIGGER [comcast_dba_sysjobs_insert_log_entry]
   ON  dbo.sysjobs
   AFTER INSERT
AS
BEGIN
 -- SET NOCOUNT ON added to prevent extra result sets from
 -- interfering with SELECT statements.
 SET NOCOUNT ON;

    -- Insert statements for trigger here
 IF EXISTS (select 1 from DBA.sys.objects where name = ''Logging_msdb_sysjobs_modifications'')
  BEGIN
   INSERT INTO [DBA].[dbo].[Logging_msdb_sysjobs_modifications] (job_id, originating_server_id, name, enabled, description, start_step_id, category_id, owner_sid, notify_level_eventlog, notify_level_email, notify_level_netsend, notify_level_page, notify_email_operator_id, notify_netsend_operator_id, notify_page_operator_id, delete_level, date_created, date_modified, version_number, host_name, login_name, modification_type, insert_date_time)
   SELECT job_id, originating_server_id, name, enabled, description, start_step_id, category_id, owner_sid, notify_level_eventlog, notify_level_email, notify_level_netsend, notify_level_page, notify_email_operator_id, notify_netsend_operator_id, notify_page_operator_id, delete_level, date_created, date_modified, version_number, host_name(), suser_sname(), ''INSERT'', GETDATE()
   FROM inserted i
  END

END')

EXEC('CREATE TRIGGER [comcast_dba_sysjobsteps_update_log_entry]
   ON  dbo.sysjobsteps
   AFTER UPDATE
AS
BEGIN
 -- SET NOCOUNT ON added to prevent extra result sets from
 -- interfering with SELECT statements.
 SET NOCOUNT ON;

    /*only log updates where the job step command is different */
 /*IF (SELECT command from inserted) <> (select command from deleted) or (SELECT step_name from inserted) <> (select step_name from deleted)*/
 IF EXISTS
 (
  SELECT 1
  FROM inserted i
  inner join deleted d
  on i.step_uid = d.step_uid
  where i.command <> d.command
  or i.step_name <> d.step_name
  or i.step_id <> d.step_id
  )
 BEGIN

  IF EXISTS (select 1 from DBA.sys.objects where name = ''Logging_msdb_sysjobsteps_modifications'')
  BEGIN

   INSERT INTO [DBA].[dbo].[Logging_msdb_sysjobsteps_modifications] (job_id, step_id, step_name, subsystem, command, flags, cmdexec_success_code, on_success_action, on_success_step_id, on_fail_action, on_fail_step_id, server, database_name, database_user_name, retry_attempts, retry_interval, os_run_priority, output_file_name, last_run_outcome, last_run_duration, last_run_retries, last_run_date, last_run_time, proxy_id, step_uid, host_name, login_name, modification_type, insert_date_time)
