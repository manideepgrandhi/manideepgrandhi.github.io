resource database details

resource
--query resource database version and last update time
select SERVERPROPERTY('resourceversion')
select SERVERPROPERTY('ResourceLastUpdateDateTime')
--Location of resource database files
Use master
GO
SELECT 'ResourceDB' AS 'Database Name'
    , NAME AS [Database File]
    , FILENAME AS [Database File Location]
FROM sys.sysaltfiles
    WHERE DBID = 32767
GO
--attach resource database as new database using attach method
USE [master]
GO
CREATE DATABASE [mssqlsystemresource1] ON
( FILENAME = N'E:\SQL\Data\mssqlsystemresource.mdf' ),
( FILENAME = N'E:\SQL\Data\mssqlsystemresource.ldf' )
 FOR ATTACH
GO
