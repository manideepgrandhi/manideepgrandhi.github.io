Automation-SQL server Restart email.

Automation
USE [msdb]
GO

/****** Object:  Job [DBA - Alert - SQL Agent Started]    Script Date: 3/4/2016 6:29:50 AM ******/
BEGIN TRANSACTION
DECLARE @ReturnCode INT
SELECT @ReturnCode = 0
/****** Object:  JobCategory [DBA Alerts]    Script Date: 3/4/2016 6:29:50 AM ******/
IF NOT EXISTS (SELECT name FROM msdb.dbo.syscategories WHERE name=N'DBA Alerts' AND category_class=1)
BEGIN
EXEC @ReturnCode = msdb.dbo.sp_add_category @class=N'JOB', @type=N'LOCAL', @name=N'DBA Alerts'
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback

END

DECLARE @jobId BINARY(16)
EXEC @ReturnCode =  msdb.dbo.sp_add_job @job_name=N'DBA - Alert - SQL Agent Started',
  @enabled=1,
  @notify_level_eventlog=2,
  @notify_level_email=0,
  @notify_level_netsend=0,
  @notify_level_page=0,
  @delete_level=0,
  @description=N'No description available.',
  @category_name=N'DBA Alerts',
  @owner_login_name=N'sa', @job_id = @jobId OUTPUT
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
/****** Object:  Step [Send Alert Email]    Script Date: 3/4/2016 6:29:50 AM ******/
EXEC @ReturnCode = msdb.dbo.sp_add_jobstep @job_id=@jobId, @step_name=N'Send Alert Email',
  @step_id=1,
  @cmdexec_success_code=0,
  @on_success_action=3,
  @on_success_step_id=2,
  @on_fail_action=2,
  @on_fail_step_id=2,
  @retry_attempts=0,
  @retry_interval=1,
  @os_run_priority=0, @subsystem=N'TSQL',
  @command=N'DECLARE @v_recipients VARCHAR(255),
  @v_body VARCHAR(1000),
  @v_subject VARCHAR(255),
  @v_profile_name VARCHAR(255),
  @v_importance VARCHAR(6),
  @v_body_format VARCHAR (20)

SET @v_recipients = ''_chq--ie-infrasupport_@abc.com''; --semicolon delimited list
SET @v_body = ''The SQL Server Agent was started on ''+@@servername+''.  Possible causes: server reboot, failover, or agent was not running and was turned on.''
SET @v_subject = ''Non-Production: ''+@@servername+'': ALERT: SQL Agent Started''
SET @v_profile_name = ''IE_SQL_DBA_Profile''
SET @v_importance = ''High'' --Low, Normal, High
SET @v_body_format = ''TEXT'' --TEXT, HTML

EXECUTE msdb.dbo.sp_send_dbmail
   @recipients=@v_recipients,
   @body= @v_body,
   @subject = @v_subject,
   @profile_name = @v_profile_name,
   @importance = @v_importance,
   @body_format = @v_body_format',
  @database_name=N'master',
  @flags=0
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
/****** Object:  Step [Insert Into DBA Alerts and Warnings Table]    Script Date: 3/4/2016 6:29:50 AM ******/
EXEC @ReturnCode = msdb.dbo.sp_add_jobstep @job_id=@jobId, @step_name=N'Insert Into DBA Alerts and Warnings Table',
  @step_id=2,
  @cmdexec_success_code=0,
  @on_success_action=1,
  @on_success_step_id=0,
  @on_fail_action=2,
  @on_fail_step_id=0,
  @retry_attempts=0,
  @retry_interval=0,
  @os_run_priority=0, @subsystem=N'TSQL',
  @command=N'DECLARE @server_name varchar(255),
  @job_name varchar(100)

SELECT @server_name = CONVERT(varchar(255), @@servername)

SELECT @job_name = convert(varchar(100), [name])
FROM msdb.dbo.sysjobs
WHERE job_id = CONVERT(uniqueidentifier, $(ESCAPE_NONE(JOBID)))

INSERT INTO [dbo].[DBA_Alerts_Warnings] ([alert_description]) VALUES (@job_name)

',
  @database_name=N'DBA',
  @flags=0
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_update_job @job_id = @jobId, @start_step_id = 1
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_add_jobschedule @job_id=@jobId, @name=N'On Agent Start',
  @enabled=1,
  @freq_type=64,
  @freq_interval=0,
  @freq_subday_type=0,
  @freq_subday_interval=0,
  @freq_relative_interval=0,
  @freq_recurrence_factor=0,
  @active_start_date=20080410,
  @active_end_date=99991231,
  @active_start_time=0,
  @active_end_time=235959,
  @schedule_uid=N'75a9ea1c-98e2-492c-85a1-3f328ed8ff79'
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_add_jobserver @job_id = @jobId, @server_name = N'(local)'
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
COMMIT TRANSACTION
GOTO EndSave
QuitWithRollback:
    IF (@@TRANCOUNT > 0) ROLLBACK TRANSACTION
EndSave:

GO
