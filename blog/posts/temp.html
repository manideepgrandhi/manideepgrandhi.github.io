SELECT deqs.query_hash, deqs.query_plan_hash, deqp.query_plan
FROM sys.dm_exec_query_stats AS deqs
CROSS APPLY sys.dm_exec_query_plan(deqs.plan_handle) AS deqp
WHERE deqs.query_plan_hash = 0x00ED538D26FCD990
OPTION(RECOMPILE)
