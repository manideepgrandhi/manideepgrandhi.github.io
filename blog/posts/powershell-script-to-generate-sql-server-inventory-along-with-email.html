Powershell Script to generate SQL Server inventory along with email

powershell

<###########################################################################

#
#
#https://msdn.microsoft.com/en-us/library/microsoft.sqlserver.management.smo.server.aspx
#
###########################################################################>

param(
    [string]$InputFile,
    [string]$OutputFile
)

#$ServerListFile = $InputFile

#$OutputFile = $OutputFile

#$ServerList = Get-Content $ServerListFile -ErrorAction SilentlyContinue
$ServerList = Get-Content "C:\Users\ptmp001c\srvdb.txt"
$OutputFile="C:\Users\ptmp001c\dbinventory_$(get-date -f yyyy-MM-dd).htm"

[Reflection.Assembly]::LoadWithPartialName("Microsoft.SqlServer.Smo")
$Result = @()
Foreach($SQLServer in $ServerList)
{
    $SQLServerInfo = new-object “Microsoft.SqlServer.Management.Smo.Server” $SQLServer
    $Result += $SQLServerInfo | Select Name,DefaultFile,DefaultLog, Edition,ProductLevel, Version,ErrorLogPath,FilestreamLevel, ServiceAccount,ServerType, Platform, IsClustered, PhysicalMemory, Processors,AvailabilityGroups,ClusterQuorumType
}

if($Result -ne $null)
{
    $HTML = '<style type="text/css">
    #Header{font-family:"Trebuchet MS", Arial, Helvetica, sans-serif;width:100%;border-collapse:collapse;}
    #Header td, #Header th {font-size:14px;border:1px solid #98bf21;padding:3px 7px 2px 7px;}
    #Header th {font-size:14px;text-align:left;padding-top:5px;padding-bottom:4px;background-color:#A7C942;color:#fff;}
    #Header tr.alt td {color:#000;background-color:#EAF2D3;}
    </Style>'
    $HTML += "<HTML><BODY><Table border=1 cellpadding=0 cellspacing=0 width=100% id=Header>

        <TR>

            <TH><B>Server Name</B></TH>

            <TH><B>PhysicalMemory (MB)</B></TH>

            <TH><B>Logical Processors</B></TH>

            <TH><B>ServerType</B></TH>

            <TH><B>Platform</B></TH>

            <TH><B>DefaultFile</B></TH>

            <TH><B>DefaultLog</B></TH>

            <TH><B>Edition</B></TD>

            <TH><B>ProductLevel</B></TH>

            <TH><B>Version</B></TH>

            <TH><B>ErrorLogPath</B></TH>

            <TH><B>FilestreamLevel</B></TH>

            <TH><B>ServiceAccount</B></TH>

            <TH><B>IsClustered</B></TH>

            <TH><B>AvailabilityGroups</B></TH>

            <TH><B>ClusterQuorumType</B></TH>

        </TR>"

    Foreach($Entry in $Result)
    {
        $HTML += "<TR>
                        <TD>$($Entry.Name)</TD>
                        <TD align=center>$($Entry.PhysicalMemory)</TD>
                        <TD align=center>$($Entry.Processors)</TD>
                        <TD>$($Entry.ServerType)</TD>
                        <TD>$($Entry.Platform)</TD>
                        <TD>$($Entry.DefaultFile)</TD>
                        <TD>$($Entry.DefaultLog)</TD>
                        <TD>$($Entry.Edition)</TD>
                        <TD align=center>$($Entry.ProductLevel)</TD>
                        <TD>$($Entry.Version)</TD>
                        <TD>$($Entry.ErrorLogPath)</TD>
                        <TD>$($Entry.FilestreamLevel)</TD>
                        <TD>$($Entry.ServiceAccount)</TD>
                        <TD align=center>$($Entry.IsClustered)</TD>
                        <TD>$($Entry.AvailabilityGroups)</TD>
                        <TD>$($Entry.ClusterQuorumType)</TD>
                    </TR>"
    }
    $HTML += "</Table></BODY></HTML>"

    $HTML | Out-File $OutputFile
}

$body = [System.IO.File]::ReadAllText("C:\Users\ptmp001c\dbinventory_$(get-date -f yyyy-MM-dd).htm")
Send-MailMessage -To Manideep_Grandhi@xyz.com -From IE_Infra_Monitor@xyz.com -Subject "Database Inventory Report $(get-date -f yyyy-MM-dd) " -Body $body -BodyAsHtml -Priority High -SmtpServer mailrelay.xyz.com
