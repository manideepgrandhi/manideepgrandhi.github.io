--2012-2014 db ddl trigger

--CREATE trigger with the following items to log
CREATE TRIGGER [dba_trigger_log_db_events]
   ON  DATABASE

   FOR GRANT_DATABASE,  --GRANT DB rights
  DENY_DATABASE,   --DENY db rights
  REVOKE_DATABASE, --revoke DB rights
  CREATE_FUNCTION, --create functions
  ALTER_FUNCTION,  --alter functions
  DROP_FUNCTION,   --drop functions
  CREATE_INDEX,    --create indexes
  ALTER_INDEX,  --alter indexes
  DROP_INDEX,   --drop indexes
  CREATE_PARTITION_FUNCTION, --file partitions?
  ALTER_PARTITION_FUNCTION, --file partitions?
  DROP_PARTITION_FUNCTION, --file partitions?
  CREATE_PROCEDURE, --create SPs
  ALTER_PROCEDURE,  --alter SPs
  DROP_PROCEDURE,  --DROP SPs
  CREATE_ROLE, --create DB role
  ALTER_ROLE, --modify DB role
  DROP_ROLE, --drop DB role
  CREATE_SCHEMA, --create DB schema
  ALTER_SCHEMA, --alter DB schema
  DROP_SCHEMA, --drop DB schema
  CREATE_STATISTICS, --stats
  DROP_STATISTICS, --stats
  UPDATE_STATISTICS, --stats
  CREATE_TABLE, --new table
  ALTER_TABLE, --alter table
  DROP_TABLE, --drop table
  CREATE_TRIGGER, --new trigger
  ALTER_TRIGGER,  --alter trigger
  DROP_TRIGGER,  --drop trigger
  CREATE_USER, --new db user
  ALTER_USER, --alter db user
  DROP_USER, --drop db user
  CREATE_VIEW, --new view
  ALTER_VIEW, --alter view
  DROP_VIEW, --drop view
  CREATE_XML_SCHEMA_COLLECTION, --XML objects
  ALTER_XML_SCHEMA_COLLECTION,
  DROP_XML_SCHEMA_COLLECTION
AS

--GET event data through system function
declare @data xml
 set @data = EVENTDATA()

set arithabort on
set ansi_padding on
set ansi_warnings on
set CONCAT_NULL_YIELDS_NULL on

--pull free space from DBA DB
declare @dba_db_available_space_mb decimal(18,2)

EXEC sp_executesql N'USE DBA;
SELECT    @dba_db_available_space_mb_output = CONVERT(DECIMAL(18,2),SUM(size/128.0 - CAST(FILEPROPERTY(name, ''SpaceUsed'') AS int)/128.0))
FROM sys.database_files
WHERE type_desc = ''ROWS''',
N'@dba_db_available_space_mb_output decimal(18,2) OUTPUT',
@dba_db_available_space_mb_output = @dba_db_available_space_mb OUTPUT

/*
SELECT @dba_db_available_space_mb,
    @dba_db_available_space_mb*1024.0 [dba_db_available_space_kb]
*/

IF EXISTS (select [name] from DBA.sys.tables where name = 'DBA_DB_DDL_Trigger_Event_Information')
BEGIN
    IF @dba_db_available_space_mb > 0.02
    BEGIN

        --pull values from the event data
        INSERT INTO [DBA].[dbo].[DBA_DB_DDL_Trigger_Event_Information]
         (
          [EventType],
          [DatabaseName],
          [ServerName],
          [LoginName],
          [UserName],
          [SchemaName],
          [ObjectName],
          [ObjectType],
          [TSQLCommand],
          [HostName],
          [AppName]
         )
        SELECT
          ISNULL(@data.value('(/EVENT_INSTANCE/EventType)[1]', 'varchar(100)'), 'N/A') [EventType] ,
          ISNULL(@data.value('(/EVENT_INSTANCE/DatabaseName)[1]', 'varchar(256)'), 'N/A') [DatabaseName],
          ISNULL(@data.value('(/EVENT_INSTANCE/ServerName)[1]', 'varchar(256)'), 'N/A') [ServerName],
          ISNULL(@data.value('(/EVENT_INSTANCE/LoginName)[1]', 'varchar(1000)'), 'N/A') [LoginName],
          ISNULL(@data.value('(/EVENT_INSTANCE/UserName)[1]', 'varchar(1000)'), 'N/A') [UserName],
          ISNULL(@data.value('(/EVENT_INSTANCE/SchemaName)[1]', 'varchar(256)'), 'N/A') [SchemaName],
          ISNULL(@data.value('(/EVENT_INSTANCE/ObjectName)[1]', 'varchar(256)'), 'N/A') [ObjectName],
          ISNULL(@data.value('(/EVENT_INSTANCE/ObjectType)[1]', 'varchar(256)'), 'N/A') [ObjectType],
          ISNULL(@data.value('(/EVENT_INSTANCE/TSQLCommand)[1]', 'varchar(8000)'), 'N/A') [TSQLCommand],
          ISNULL(Host_Name(), 'N/A') [HostName],
          ISNULL(App_Name(), 'N/A') [AppName]
    END
END

GO

SET ANSI_NULLS OFF
GO

SET QUOTED_IDENTIFIER OFF
GO
