script to kill connections to a specific database when the number of connections to that database are greater than @max_conn.

DECLARE @Database_name_var varchar(255)
DECLARE @loginame_var varchar(255)
DECLARE @CMDKILL varchar(255)
DECLARE @SPID smallint
declare @counter int
declare @max_conn int
declare @seconds_to_run int
-- CHANGE THESE VALUES BEFORE RUNNING
SELECT @Database_name_var = 'Infra'
SELECT @loginame_var ='EAGLE\INFRADBPROD'
select @max_conn = 275
select @seconds_to_run = 1200

select @counter=0
while @counter < @seconds_to_run
begin
--Two loops exist.  The inner loop will kill current connections.
--  The outer loop will detect if new connections have been established which
--  then need to be killed by the inner loop.
IF (SELECT COUNT(spid) FROM master..sysprocesses WHERE loginame = @loginame_var and db_name(dbid) = @Database_name_var) > @max_conn
BEGIN
  WHILE (SELECT COUNT(spid) FROM master..sysprocesses WHERE loginame = @loginame_var and db_name(dbid) = @Database_name_var) > @max_conn
   BEGIN
      SELECT @spid = (SELECT top 1 spid FROM master..sysprocesses WHERE loginame = @loginame_var and  db_name(dbid) = @Database_name_var)
      SELECT @cmdkill = 'KILL ' + CONVERT(VARCHAR(5),@spid)
      EXEC (@cmdkill)
      --This Delay exists to allow SQL Server time to clean up the connection.
      WAITFOR DELAY '00:00:01'
   END
END
WAITFOR DELAY '00:00:01'
select @counter=@counter+1
end
