Database users without login

security
SELECT
CASE
WHEN DATALENGTH(sid) = 28 AND type = 'S'
-- only want SQL users AND principal_id > 4
-- ignore built in users THEN 1
ELSE 0
END AS is_user_without_login,*
FROM sys.database_principals
where name in ('bjakka')
