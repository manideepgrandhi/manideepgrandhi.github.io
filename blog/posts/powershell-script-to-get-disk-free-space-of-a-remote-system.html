powershell script to get disk free space of a remote system

powershell
$TotalGB = @{Name="Capacity(GB)";expression={[math]::round(($_.Capacity/ 1073741824),2)}}
$FreeGB = @{Name="FreeSpace(GB)";expression={[math]::round(($_.FreeSpace / 1073741824),2)}}
$FreePerc = @{Name="Free(%)";expression={[math]::round(((($_.FreeSpace / 1073741824)/($_.Capacity / 1073741824)) * 100),0)}}

Function Show-DriveSizes ([string]$Server)
{
 $Date = Get-Date
                            Write-Host -foregroundcolor DarkBlue -backgroundcolor yellow "$Server - - $Date"
                            $volumes = Get-WmiObject -computer $Server win32_volume
                            #| Where-object {$_.DriveLetter -eq $null}
$volumes | Select SystemName, name,Label, $TotalGB, $FreeGB, $FreePerc | Format-Table -AutoSize
}
