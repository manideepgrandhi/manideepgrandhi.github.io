DROP TABLE [dbo].[DBA_Monitor_Processes]

CREATE TABLE [dbo].[DBA_Monitor_Processes](
    [Time] [datetime] NULL,
    [SessionId] [int] NULL,
    [HostName] [nvarchar](50) NULL,
    [LoginName] [nvarchar](50) NULL,
    [NTUserName] [nvarchar](50) NULL,
    [Session_Status] [nvarchar](50) NULL,
    [Request_Status] [nvarchar](50) NULL,
    [WaitType] [nvarchar](50) NULL,
    [Wait(ms)] [int] NULL,

    [Object] [nvarchar](max) NULL,
    [StepText] [nvarchar](max) NULL
) ON [PRIMARY]

    [Time] ASC
)WITH (PAD_INDEX  = OFF, STATISTICS_NORECOMPUTE  = OFF, SORT_IN_TEMPDB = OFF, IGNORE_DUP_KEY = OFF, DROP_EXISTING = OFF, ONLINE = OFF, ALLOW_ROW_LOCKS  = ON, ALLOW_PAGE_LOCKS  = ON, FILLFACTOR = 85) ON [PRIMARY];

/*** CREATE PROCEDURE TO QUERY FOR THE DATA ***/

SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO

IF  EXISTS (SELECT * FROM sys.objects WHERE object_id = OBJECT_ID(N'[dbo].[ap_DBA_get_current_processes]') AND type in (N'P'))
DROP PROCEDURE [dbo].[ap_DBA_get_current_processes]
GO

--
CREATE PROCEDURE [dbo].[ap_DBA_get_current_processes]

AS
BEGIN

SET NOCOUNT ON;

select    getdate() time,
        s.session_id,
        s.host_name,
        s.login_name,
        s.nt_user_name,
        s.status,
        qs.status,
        qs.wait_type,
        qs.wait_time as [wait ms],
        qs.wait_resource,
        qs.blocking_session_id as [blocked],
        qs.total_elapsed_time as [process ms],
        cast(86400000*cast(getdate()-s.last_request_start_time as real) as bigint) [elapsed msec],
        db_name(qs.database_id) as db_name,
        case when st.objectid is null then st.text else db_name(st.dbid) + '.' + object_schema_name(st.objectid,st.dbid) + '.' + object_name(st.objectid,st.dbid) end as [object],

        SUBSTRING(st.text,(qs.statement_start_offset/2)+1,((CASE qs.statement_end_offset WHEN -1 THEN DATALENGTH(st.text) ELSE qs.statement_end_offset END - qs.statement_start_offset)/2) + 1) AS [step text]

from
    master.sys.dm_exec_sessions s
        left join
    master.sys.dm_exec_requests qs
        on s.session_id = qs.session_id
    OUTER APPLY sys.dm_exec_sql_text(qs.sql_handle) AS st
where (

        s.host_name is not null -- make sure we know what host connection is coming from

        and
        (
            --check for statements that are currently executing that are either
            --    a) waiting but not in the waitfor wait type
            --  b) not waiting, not sleeping, running longer than 3sec
            qs.wait_type != 'WAITFOR'
            or
            (
                --no waiting, spid is not sleeping, and total elapsed time is > 3sec
                qs.wait_type is null
                and
                s.status!='sleeping'
                and qs.total_elapsed_time>3000
             )
         )
       )
       -- pull in all blocking
       or s.session_id in (select blocking_session_id from master.sys.dm_exec_requests)
order by qs.total_elapsed_time desc

END

GO

/*** CREATE JOB ***/

USE [msdb]
GO

/****** Object:  Job [DBA_Monitor_Processes]    Script Date: 04/27/2010 16:08:42 ******/
IF  EXISTS (SELECT job_id FROM msdb.dbo.sysjobs_view WHERE name = N'DBA_Monitor_Processes')
EXEC msdb.dbo.sp_delete_job @job_name=N'DBA_Monitor_Processes', @delete_unused_schedule=1
GO

USE [msdb]
GO

/****** Object:  Job [DBA_Monitor_Processes]    Script Date: 04/27/2010 16:08:42 ******/
BEGIN TRANSACTION
DECLARE @ReturnCode INT
SELECT @ReturnCode = 0
/****** Object:  JobCategory [[Uncategorized (Local)]]]    Script Date: 04/27/2010 16:08:42 ******/
IF NOT EXISTS (SELECT name FROM msdb.dbo.syscategories WHERE name=N'[Uncategorized (Local)]' AND category_class=1)
BEGIN
EXEC @ReturnCode = msdb.dbo.sp_add_category @class=N'JOB', @type=N'LOCAL', @name=N'[Uncategorized (Local)]'
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback

END

DECLARE @jobId BINARY(16)
EXEC @ReturnCode =  msdb.dbo.sp_add_job @job_name=N'DBA_Monitor_Processes',
        @enabled=1,
        @notify_level_eventlog=0,
        @notify_level_email=0,
        @notify_level_netsend=0,
        @notify_level_page=0,
        @delete_level=0,
        @description=N'No description available.',
        @category_name=N'[Uncategorized (Local)]',
        @owner_login_name=N'sa', @job_id = @jobId OUTPUT
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
/****** Object:  Step [Exec  Insert Statement]    Script Date: 04/27/2010 16:08:43 ******/
EXEC @ReturnCode = msdb.dbo.sp_add_jobstep @job_id=@jobId, @step_name=N'Exec  Insert Statement',
        @step_id=1,
        @cmdexec_success_code=0,
        @on_success_action=3,
        @on_success_step_id=0,
        @on_fail_action=2,
        @on_fail_step_id=0,
        @retry_attempts=0,
        @retry_interval=0,
        @os_run_priority=0, @subsystem=N'TSQL',
        @command=N'Insert into [dbo].[DBA_Monitor_Processes] exec [dbo].[ap_DBA_get_current_processes] ',
        @database_name=N'DBA',
        @flags=0
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
/****** Object:  Step [purge rolling 90 days]    Script Date: 04/27/2010 16:08:43 ******/
EXEC @ReturnCode = msdb.dbo.sp_add_jobstep @job_id=@jobId, @step_name=N'purge rolling 90 days',
        @step_id=2,
        @cmdexec_success_code=0,
        @on_success_action=1,
        @on_success_step_id=0,
        @on_fail_action=2,
        @on_fail_step_id=0,
        @retry_attempts=0,
        @retry_interval=0,
        @os_run_priority=0, @subsystem=N'TSQL',
        @command=N'delete
  FROM [DBA].[dbo].[DBA_Monitor_Processes]
WHERE [Time] < dateadd(d, -90, getdate())',
        @database_name=N'DBA',
        @flags=0
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_update_job @job_id = @jobId, @start_step_id = 1
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_add_jobschedule @job_id=@jobId, @name=N'Every 10 sec',
        @enabled=1,
        @freq_type=4,
        @freq_interval=1,
        @freq_subday_type=2,
        @freq_subday_interval=10,
        @freq_relative_interval=0,
        @freq_recurrence_factor=0,
        @active_start_date=20090711,
        @active_end_date=99991231,
        @active_start_time=0,
        @active_end_time=235959
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
EXEC @ReturnCode = msdb.dbo.sp_add_jobserver @job_id = @jobId, @server_name = N'(local)'
IF (@@ERROR <> 0 OR @ReturnCode <> 0) GOTO QuitWithRollback
COMMIT TRANSACTION
GOTO EndSave
QuitWithRollback:
    IF (@@TRANCOUNT > 0) ROLLBACK TRANSACTION
EndSave:

GO
